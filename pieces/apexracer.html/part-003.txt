function updateCam(dt) {
  const P = R.P, T = R.T, sp = Math.abs(P.spd), fx = Math.sin(P.h), fz = Math.cos(P.h), y = T.y;
  const vh = sp > 3 ? Math.atan2(P.vx, P.vz) : P.h; let dh = vh - P.h; while (dh > Math.PI) dh -= TAU; while (dh < -Math.PI) dh += TAU; const ch = P.h + dh * 0.45;
  const cx = Math.sin(ch), cz = Math.cos(ch);
  const pre = R.t < -1.5 ? clamp((-R.t - 1.5) / 2.7, 0, 1) : 0;
  let back, up, look;
  if (R.camMode === 2 && !P.fin) { back = -0.6; up = 1.25; look = 30; }
  else if (R.camMode === 1) { back = 10 + sp * 0.03; up = 3.8; look = 6; }
  else { back = 6.2 + sp * 0.022; up = 2.25 + sp * 0.004; look = 5; }
  if (P.fin) { const a = R.t * 0.35; back = 9; up = 3; cam.position.lerp(new THREE.Vector3(P.x + Math.sin(P.h + 2.4 + a) * back, y + up, P.z + Math.cos(P.h + 2.4 + a) * back), 1 - Math.exp(-3 * dt)); cam.lookAt(P.x, y + 0.9, P.z); }
  else {
    const want = new THREE.Vector3(P.x - cx * back + pre * (-cx * 22 + cz * 8), y + up + pre * 14, P.z - cz * back + pre * (-cz * 22 - cx * 8));
    if (R.camMode === 2) cam.position.copy(want); else cam.position.lerp(want, 1 - Math.exp(-(R.t < 0 ? 3 : 9) * dt));
    cam.lookAt(P.x + fx * look, y + (R.camMode === 2 ? 1.1 : 1.0), P.z + fz * look);
  }
  if (R.shake > 0) { cam.position.x += (Math.random() - 0.5) * R.shake * 0.5; cam.position.y += (Math.random() - 0.5) * R.shake * 0.4; R.shake = Math.max(0, R.shake - dt * 2.5); }
  const fov = 60 + Math.min(26, sp * 0.26) + (P.nosOn ? 9 : 0); R.fov += (fov - R.fov) * Math.min(1, dt * 4); cam.fov = R.fov; cam.updateProjectionMatrix();
  post.spd = clamp((sp - 30) / 60, 0, 1) * 0.022 + (P.nosOn ? 0.018 : 0); post.ca = P.nosOn ? 0.012 : sp > 50 ? 0.005 : 0;
  const W = R.W; if (W.sky) W.sky.position.copy(cam.position);
  if (W.sun.castShadow) { W.sun.position.set(P.x + W.sunD.x * 120, y + W.sunD.y * 120, P.z + W.sunD.z * 120); W.sun.target.position.set(P.x, y, P.z); }
  if (W.head) { W.head.position.set(P.x + fx * 1.5, y + 1.1, P.z + fz * 1.5); W.head.target.position.set(P.x + fx * 30, y, P.z + fz * 30); }
}

function playerRating() { return Math.max(1, S.wins * 2 + Math.floor(S.rep / 500)); }
function friendCode() {
  const c = S.cars[S.cur] || defCfg('vanta');
  const j = JSON.stringify({ n: (user || 'Racer').slice(0, 16), r: playerRating(), car: S.cur, p: c.paint, f: c.fin, ri: c.rim || 0, g: c.glow || '' });
  const b = btoa(unescape(encodeURIComponent(j))).replace(/\+/g, '~').replace(/\//g, '_').replace(/=+$/, '');
  return 'RACE-' + (b.match(/.{1,4}/g) || [b]).join('-');
}
function parseFriendCode(s) {
  s = String(s || '').trim(); if (!/^RACE-/i.test(s)) return null;
  let b = s.slice(5).replace(/-/g, '').replace(/~/g, '+').replace(/_/g, '/');
  if (!b || /[^A-Za-z0-9+\/]/.test(b)) return null;
  while (b.length % 4) b += '=';
  try {
    const o = JSON.parse(decodeURIComponent(escape(atob(b))));
    if (!o || typeof o.n !== 'string') return null;
    return { n: String(o.n).slice(0, 16) || 'Rival', r: clamp(Math.round(+o.r || 1), 1, 999), car: CAR[o.car] ? o.car : 'vanta',
      p: /^#[0-9a-fA-F]{6}$/.test(o.p) ? o.p : null, f: FIN[o.f] ? o.f : 'metal', ri: clamp(Math.round(+o.ri || 0), 0, 5), g: typeof o.g === 'string' ? o.g : '' };
  } catch (e) { return null; }
}
