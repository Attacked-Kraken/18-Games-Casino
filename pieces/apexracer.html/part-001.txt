function detailGeo(pk, paintHex) {
  const key = 'det_' + pk + '_' + paintHex; if (GC[key]) return GC[key];
  const P = PROF[pk], ch = P.ch, L2 = P.L / 2, L = [];
  const add = (geo, x, y, z, c, rx, ry, rz, sx, sy, sz) => L.push([geo, M4(x, y, z, rx || 0, ry || 0, rz || 0, sx || 1, sy || 1, sz || 1), c]);
  const D = DETC.dark, CB = DETC.carbon, CH = DETC.chrome;
  const cyl = (rt, rb, h, sg) => new THREE.CylinderGeometry(rt, rb, h, sg || 14);
  const iy = P.clr + 0.14;
  /* nose intake: dark mouth sunk into the bumper + corner pods, all rooted */
  const noseIntake = (wf, divider) => { const w = P.W * wf;
    add(BOXG, L2 - 0.05, iy, 0, D, 0, 0, 0, 0.2, 0.15, w);
    if (divider) add(BOXG, L2 + 0.03, iy, 0, CH, 0, 0, 0, 0.05, 0.035, w * 0.9);
    for (const s of [-1, 1]) { const pz = sideZ(P, L2 - 0.06, iy) - 0.13;
      add(BOXG, L2 - 0.07, iy + 0.01, s * pz, D, 0, s * 0.35, 0, 0.18, 0.12, 0.24); } };
  /* hood vents: thin dark insets lying on the hood surface */
  const hoodVent = (vx, vz, w, rot) => { for (const s of [-1, 1])
    add(BOXG, vx, topY(P, vx) + 0.004, s * vz, D, 0, s * 0.1, rot || 0, w, 0.02, 0.1); };
  /* side intake: dark recess sunk into the flank + vertical slats at the mouth */
  const sideIntake = (ix, iyy, w, h, slats, rot) => { for (const s of [-1, 1]) {
    const iz = sideZ(P, ix, iyy), d = 0.1;
    add(BOXG, ix, iyy, s * (iz - 0.02 - d / 2), D, 0, s * 0.05, rot || 0, w, h, d);
    for (let i = 0; i < slats; i++) { const t = slats === 1 ? 0 : (i / (slats - 1) - 0.5);
      add(BOXG, ix + t * w * 0.3, iyy, s * (iz - 0.03), CB, 0, s * 0.05, rot || 0, 0.045, h * 0.92, 0.05); } } };
  /* front canards: dive planes rooted in the nose corners */
  const canards = () => { for (const s of [-1, 1]) { const cz = sideZ(P, L2 - 0.22, P.clr + 0.2) - 0.06;
    add(BOXG, L2 - 0.24, P.clr + 0.2, s * cz, CB, 0, s * 0.25, 0.22, 0.26, 0.025, 0.16); } };
  /* arch flares: torus lips hugging the wheel-arch cutouts */
  const archFlares = (tube) => { for (const s of [-1, 1]) for (const wx of [P.fx, P.rx])
    add(new THREE.TorusGeometry(P.r + 0.085, tube, 6, 22, Math.PI), wx, P.r + 0.02, s * (P.W / 2 - 0.045), D, 0, 0, 0); };

  const roofScoop = () => { const sx = 0.72, sy = topY(P, sx);

    add(BOXG, -0.28, 0.78, s * (sideZ(P, -0.28, 0.78) - 0.012), D, 0, 0, 0, 0.17, 0.032, 0.05);

    add(BOXG, anx, topY(P, anx) + 0.035, 0, D, 0, 0, -0.28, 0.17, 0.1, 0.032); }
  /* interior silhouettes, kept under the canopy line */
  const cp = Math.max(...P.cab.map((p) => p[1]));

  /* per-car character */
  if (ch === 'van') {

    for (let i = 0; i < 3; i++) add(BOXG, L2 + 0.02, 0.66 + i * 0.12, 0, CH, 0, 0, 0, 0.04, 0.035, P.W * 0.52);
    for (const s of [-1, 1]) {
      add(BOXG, -0.6, 1.15, s * (sideZ(P, -0.6, 1.15) - 0.008), D, 0, 0, 0, 1.9, 0.035, 0.03);

  } else if (ch === 'gt') { noseIntake(0.52, true); hoodVent(1.3, 0.3, 0.34); sideIntake(-0.55, 0.66, 0.55, 0.24, 1); archFlares(0.04);
  } else if (ch === 'track') { noseIntake(0.6, false); hoodVent(1.25, 0.3, 0.3); hoodVent(0.95, 0.3, 0.3); sideIntake(-0.55, 0.66, 0.8, 0.3, 3); canards(); archFlares(0.045); roofScoop();
  } else if (ch === 'muscle') { noseIntake(0.58, false); hoodVent(1.25, 0.32, 0.42); sideIntake(-0.55, 0.66, 0.7, 0.26, 3); canards(); archFlares(0.05);

  } else if (ch === 'longtail') { noseIntake(0.5, true);

  } else if (ch === 'widebody') { noseIntake(0.62, false); hoodVent(1.3, 0.34, 0.44); sideIntake(-0.55, 0.66, 0.85, 0.32, 1); canards(); archFlares(0.055);
  } else if (ch === 'stealth') { noseIntake(0.58, false); hoodVent(1.3, 0.3, 0.38); sideIntake(-0.55, 0.66, 0.8, 0.3, 2); canards(); archFlares(0.05);
  } else if (ch === 'sedan') { noseIntake(0.5, true); hoodVent(1.35, 0.3, 0.28); archFlares(0.03); }
  /* rear diffuser + exhausts, rooted in the bumper */
  const dyy = P.clr + 0.075;
  add(BOXG, -L2 + 0.02, dyy, 0, D, 0, 0, 0, 0.34, 0.13, P.W * (ch === 'van' ? 0.6 : 0.68));
  const fins = ch === 'track' || ch === 'stealth' || ch === 'widebody' ? 5 : 3;
  for (let i = 0; i < fins; i++) add(BOXG, -L2 - 0.02, dyy + 0.005, (i - (fins - 1) / 2) * 0.17, CB, 0, 0, 0, 0.36, 0.17, 0.024);
  if (ch === 'van') {
    for (const z of [-0.5, 0.5]) add(cyl(0.05, 0.05, 0.14, 10), -L2 - 0.03, P.clr + 0.18, z, D, 0, 0, Math.PI / 2);
  } else if (ch !== 'widebody') {
    const tips = ch === 'muscle' ? [-0.42, -0.26, 0.26, 0.42] : [-0.3, 0.3];
    for (const z of tips) { const ey = P.clr + 0.2;
      add(cyl(0.055, 0.055, 0.2, 12), -L2 - 0.02, ey, z, D, 0, 0, Math.PI / 2);
      add(cyl(0.068, 0.068, 0.06, 12), -L2 - 0.115, ey, z, CH, 0, 0, Math.PI / 2);
      add(cyl(0.048, 0.048, 0.065, 12), -L2 - 0.115, ey, z, D, 0, 0, Math.PI / 2); }
  }
  const g = merge(L); GC[key] = g; return g;
}
